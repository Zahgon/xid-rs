1//! Port of the pieces of Go's `crypto/rand` that `xid` relies on.
2//!
3//! `id.go` uses exactly one entry point:
4//!
5//! ```go
6//! if _, err := rand.Reader.Read(b); err != nil { ... }
7//! ```
8//!
9//! Go's `crypto/rand.Reader` is documented to always fill the whole buffer or
10//! return an error, so the Rust counterpart is a "fill or fail" function. The
11//! error is surfaced (not swallowed) because both call sites in `xid` panic
12//! with the error text embedded in the message.
13
14use std::io;
15
16/// Fills `b` with cryptographically secure random bytes.
17///
18/// Equivalent of Go's `rand.Reader.Read(b)` for the "read everything" contract
19/// that `crypto/rand` guarantees.
20pub fn read(b: &mut [u8]) -> io::Result<()> {
21    if b.is_empty() {
22        return Ok(());
23    }
24    fill(b)
25}
26
27#[cfg(unix)]
28fn fill(b: &mut [u8]) -> io::Result<()> {
29    use std::fs::File;
30    use std::io::Read;
31    use std::sync::{Mutex, OnceLock};
32
33    // Go keeps a single lazily-opened handle in `crypto/rand`; do the same so
34    // repeated calls do not pay for an open() each time.
35    static SOURCE: OnceLock<io::Result<Mutex<File>>> = OnceLock::new();
36
37    let source = SOURCE.get_or_init(|| File::open("/dev/urandom").map(Mutex::new));
38    match source {
39        Ok(file) => {
40            let mut guard = file.lock().unwrap_or_else(|poisoned| poisoned.into_inner());
41            guard.read_exact(b)
42        }
43        Err(e) => Err(io::Error::new(e.kind(), e.to_string())),
44    }
45}
46
47#[cfg(windows)]
48fn fill(b: &mut [u8]) -> io::Result<()> {
49    // `RtlGenRandom`, exported from advapi32 under the name `SystemFunction036`.
50    // This is what Go's `crypto/rand` used on Windows before it moved to
51    // `ProcessPrng`; both are backed by the same CSPRNG.
52    #[link(name = "advapi32")]
53    extern "system" {
54        #[link_name = "SystemFunction036"]
55        fn rtl_gen_random(random_buffer: *mut u8, random_buffer_length: u32) -> u8;
56    }
57
58    // The API takes a u32 length; chunk to stay within it.
59    for chunk in b.chunks_mut(u32::MAX as usize) {
60        let ok = unsafe { rtl_gen_random(chunk.as_mut_ptr(), chunk.len() as u32) };
61        if ok == 0 {
62            return Err(io::Error::new(io::ErrorKind::Other, "RtlGenRandom failed"));
63        }
64    }
65    Ok(())
66}
67
68#[cfg(not(any(unix, windows)))]
69fn fill(_b: &mut [u8]) -> io::Result<()> {
70    Err(io::Error::new(
71        io::ErrorKind::Unsupported,
72        "crypto/rand: no secure random source on this platform",
73    ))
