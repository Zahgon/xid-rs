1//! Port of Go's `hash/crc32.ChecksumIEEE`.
2//!
3//! Used by `init()` in `id.go` to fold the contents of `/proc/self/cpuset`
4//! into the process id when running inside a container.
5
6/// The IEEE polynomial, in reversed (LSB-first) form — Go's `crc32.IEEE`.
7const IEEE_REVERSED: u32 = 0xedb88320;
8
9/// The 256-entry lookup table Go builds with `crc32.MakeTable(crc32.IEEE)`.
10static TABLE: [u32; 256] = make_table();
11
12const fn make_table() -> [u32; 256] {
13    let mut table = [0u32; 256];
14    let mut i = 0usize;
15    while i < 256 {
16        let mut crc = i as u32;
17        let mut j = 0;
18        while j < 8 {
19            if crc & 1 == 1 {
20                crc = (crc >> 1) ^ IEEE_REVERSED;
21            } else {
22                crc >>= 1;
23            }
24            j += 1;
25        }
26        table[i] = crc;
27        i += 1;
28    }
29    table
30}
31
32/// Equivalent of Go's `crc32.ChecksumIEEE(data)`.
33pub fn checksum_ieee(data: &[u8]) -> u32 {
34    let mut crc: u32 = 0xffffffff;
35    for &b in data {
36        crc = TABLE[((crc as u8) ^ b) as usize] ^ (crc >> 8);
37    }
38    !crc
39}

41#[cfg(test)]
42mod tests {
43    use super::*;
44    use std::hint::black_box;
45
46    /// The table Go builds with `crc32.MakeTable(crc32.IEEE)`.
47    #[test]
48    fn table_matches_go() {
49        // Evaluated at runtime so the const fn itself is exercised.
50        let table = black_box(make_table());
51        assert_eq!(table, TABLE);
52        // Values published in the IEEE 802.3 reflected table.
53        assert_eq!(table[0], 0x00000000);
54        assert_eq!(table[1], 0x77073096);
55        assert_eq!(table[255], 0x2d02ef8d);
56    }
57
58    /// Incremental behaviour: the checksum only depends on the bytes.
59    #[test]
60    fn checksum_is_order_sensitive() {
61        assert_ne!(checksum_ieee(b"ab"), checksum_ieee(b"ba"));
62        assert_eq!(checksum_ieee(b"abc"), 0x352441c2);
63    }
64}
